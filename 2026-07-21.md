class Solution:
    def maxActiveSectionsAfterTrade(self, s: str) -> int:
        s = "1" + s + "1"
        segs = []
        i, n = 0, len(s)
        while i < n:
            j = i
            while j < n and s[j] == s[i]:
                j += 1
            segs.append((s[i], j - i))
            i = j

        m = len(segs)
        delta = 0
        for i in range(m):
            if i == 0 or i == m - 1:
                continue
            if segs[i][0] == "1" and segs[i - 1][0] == "0" and segs[i + 1][0] == "0":
                delta = max(delta, segs[i - 1][1] + segs[i + 1][1])
        return s.count("1") + delta - 2
